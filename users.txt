<?php
// File: users.php

$filename = "users.txt";
$message = "";

// Handle form submission
if ($_SERVER['REQUEST_METHOD'] === 'POST' && !empty($_POST['username'])) {
    $newUser = trim($_POST['username']);

    // Append new user to the file
    file_put_contents($filename, $newUser . PHP_EOL, FILE_APPEND | LOCK_EX);
    $message = "User '$newUser' added successfully!";
}

// Read users from the file
$users = [];
if (file_exists($filename)) {
    $users = file($filename, FILE_IGNORE_NEW_LINES | FILE_SKIP_EMPTY_LINES);
}
?>

<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>Users Page</title>
    <style>
        body { font-family: Arial, sans-serif; margin: 30px; }
        h1 { color: #333; }
        form { margin-bottom: 20px; }
        input[type="text"] { padding: 8px; width: 200px; }
        input[type="submit"] { padding: 8px 12px; }
        ul { list-style-type: none; padding: 0; }
        li { background: #f0f0f0; margin: 5px 0; padding: 10px; border-radius: 5px; }
        .message { color: green; margin-bottom: 15px; }
    </style>
</head>
<body>
    <h1>Users List</h1>

    <!-- Display message -->
    <?php if ($message): ?>
        <div class="message"><?php echo htmlspecialchars($message); ?></div>
    <?php endif; ?>

    <!-- Form to add new user -->
    <form method="post" action="">
        <input type="text" name="username" placeholder="Enter username" required>
        <input type="submit" value="Add User">
    </form>

    <!-- Display users -->
    <?php if (!empty($users)): ?>
        <ul>
            <?php foreach ($users as $user): ?>
                <li><?php echo htmlspecialchars($user); ?></li>
            <?php endforeach; ?>
        </ul>
    <?php else: ?>
        <p>No users found.</p>
    <?php endif; ?>
</body>
</html>
